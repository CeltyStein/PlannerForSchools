    muscleRegions.forEach(region => {
      region.addEventListener('mouseenter', (e) => {
        const muscleGroup = e.target.dataset.muscle;
        const intensity = e.target.dataset.intensity;
        const intensityLabels = ['Untrained', 'Light', 'Moderate', 'High', 'Very High', 'Intense'];
        
        tooltip.textContent = `${muscleGroup}: ${intensityLabels[intensity]}`;
        tooltip.style.display = 'block';
      });

      region.addEventListener('mousemove', (e) => {
        tooltip.style.left = e.pageX + 15 + 'px';
        tooltip.style.top = e.pageY + 15 + 'px';
      });

      region.addEventListener('mouseleave', () => {
        tooltip.style.display = 'none';
      });

      // Click to cycle through intensity levels (for demo purposes)
      region.addEventListener('click', (e) => {
        const currentIntensity = parseInt(e.target.dataset.intensity);
        const newIntensity = (currentIntensity + 1) % 6;
        updateMuscleIntensity(e.target, newIntensity);
      });
    });

    function updateMuscleIntensity(element, intensity) {
      element.dataset.intensity = intensity;
      element.className = `muscle-region intensity-${intensity}`;
    }

    // Function to set muscle intensity from your own data
    function setMuscleIntensity(muscleGroup, intensityLevel) {
      // intensityLevel should be 0-5 (0=untrained, 5=intense)
      const regions = document.querySelectorAll(`[data-muscle="${muscleGroup}"]`);
      regions.forEach(region => {
        updateMuscleIntensity(region, intensityLevel);
      });
    }

    // Example function to simulate workout data
    function simulateWorkoutData() {
      const muscles = ['Chest', 'Shoulders', 'Arms', 'Back', 'Abs', 'Legs', 'Glutes', 'Calves', 'Forearms', 'Obliques', 'Hamstrings'];
      
      muscles.forEach(muscle => {
        const randomIntensity = Math.floor(Math.random() * 6);
        setMuscleIntensity(muscle, randomIntensity);
      });
    }

    function resetHeatMap() {
      muscleRegions.forEach(region => {
        updateMuscleIntensity(region, 0);
      });
    }
